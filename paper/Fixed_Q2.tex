% ==================== PART 1: Non-Ideal Model Construction ====================
\subsection{Model Assumptions: Non-Ideal Scenario Construction}
\subsubsection{Basic Configuration}
The total mission weight is $W_0 = 1$ billion tons.
Rocket and space elevator parameters are configured as follows:
\begin{itemize}
    \item \textbf{Rocket (10 bases):} Capacity $w_R = 1500$ tons, Cost $c_R = 37000$ M\$, Time $t_R = 2.5/365$ years.
    \item \textbf{Space Elevator:} Annual capacity $C_E = 537000$ tons/year, Unit cost $60.00$ M\$/ton.
\end{itemize}

\subsubsection{Failure and Delay Modeling}
\textbf{Failure Probability}
\begin{itemize}
    \item Rocket single-transport failure probability: $p_R = 0.0016$ (derived from projected 2050 reliability).
    \item Elevator annual failures: Poisson distribution with $\lambda = 0.002$.
\end{itemize}

\textbf{Delay Time Distribution}
Failure-induced delays follow a mixed Weibull distribution:
\[
f(t) = \alpha \cdot f_{1}(t;\beta_{1},\eta_{1}) + (1 - \alpha) \cdot f_{2}(t;\beta_{2},\eta_{2})
\]
where $f_i(t) = \frac{\beta_i}{\eta_i} \left( \frac{t}{\eta_i} \right)^{\beta_i-1} e^{-(t/\eta_i)^{\beta_i}}$.
\begin{itemize}
    \item \textbf{Elevator:} $\alpha = 0.995$, $\beta_1=2.2$, $\eta_1=30/365$ (short), $\beta_2=1.5$, $\eta_2=10$ (long).
    \item \textbf{Rocket:} $\alpha = 0.9$, $\beta_1=1.5$, $\eta_1=10/365$ (short), $\beta_2=2.2$, $\eta_2=70/365$ (long).
\end{itemize}
Simulation results: average elevator delay = 3.2 years; average rocket delay = 0.2 years.

\textbf{Failure Cost Functions}
\begin{itemize}
    \item Elevator: $C_{E}(t) = \dfrac{5.175 \times 10^{7}}{1 + e^{- 0.0154(t + 19.5)(t - 13)}}$
    \item Rocket: $C_{R}(t) = \dfrac{1200}{1 + e^{- 20(t - 0.2466)}}$
\end{itemize}

\subsubsection{Simulation Framework}

    \item Mean total time: 182.2 years.

% ==================== PART 3: Summary and Recommendations ====================
\subsection{Summary and Recommended Plan}
\textbf{Performance Summary}
\begin{enumerate}
    \item \textbf{Cost:} Average 67.39 trillion USD (saves 72.7\% vs. pure rocket; 10.9\% vs. pure elevator).
    \item \textbf{Time:} Average 182.2 years (70.5\% of cases complete within 180 years).
    \item \textbf{Allocation:} Elevator handles 95.9\% of mass; rockets 4.0\%.
    \item \textbf{Failure Impact:} Delays constitute only 1.9\% of total time.
\end{enumerate}

\begin{figure}[htbp]
\centering

\caption{Joint distribution of cost and time for the transportation plan.}
\label{fig:cost_time_dist}

\textbf{Recommended Implementation Plan}
\begin{itemize}
    \item \textbf{Strategy:} Elevator-core (95.9\% mass), rocket-supplement (4.0\% mass).
    \item \textbf{Time Target:} 190 years (based on 182.2-year average with buffer).
    \item \textbf{Budget:} 78 trillion USD (based on 67.39 trillion average with contingency).
    \item \textbf{Risk Focus:} Mitigate 0.5\% probability long-delay elevator failures.
\end{itemize}

\textbf{Conclusion}
The joint transportation model demonstrates:
\begin{enumerate}
    \item Significant economic benefits over single-mode solutions.
    \item Controllable mission timeline with high completion probability.
    \item Robust reliability with minimal failure-induced delays.
    \item Optimized resource allocation leveraging elevator scale and rocket flexibility.
\end{enumerate}
This provides a feasible framework for large-scale extraterrestrial resource transportation.
